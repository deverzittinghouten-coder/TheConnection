function scheduleNextConversation(delay) { clearTimers(); typingTimer = setTimeout(() => { if (!busy) showTyping(); }, Math.max(1000, delay - 4000)); idleTimer = setTimeout(async () => { if (busy) { scheduleNextConversation(5000); return; } setBusy(true); const nextPerson = chooseParticipant(); const success = await getReply(nextPerson); setBusy(false); if (success) { // Na iedere reactie gaat het gesprek vanzelf door scheduleNextConversation(8000 + Math.random() * 11000); } else { // Bij een fout automatisch opnieuw proberen typing.textContent = ""; scheduleNextConversation(10000 + Math.random() * 5000); } }, delay); }
